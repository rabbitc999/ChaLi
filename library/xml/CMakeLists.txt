cmake_minimum_required(VERSION 3.20)
project(chali_xml LANGUAGES C)
option(CHALI_NATIVE_STATIC "Build native archives for the embedded BUILD linker" OFF)
include(FetchContent)
set(EXPAT_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(EXPAT_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(EXPAT_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(EXPAT_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(EXPAT_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(EXPAT_DTD OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(EXPAT_ARCHIVE "https://github.com/libexpat/libexpat/releases/download/R_2_10_0/expat-2.10.0.tar.gz"
    CACHE STRING "Expat release URL or local archive")
FetchContent_Declare(expat URL "${EXPAT_ARCHIVE}"
    URL_HASH SHA256=5466a341f9c692ef7a958a6881311be597f8e1ce345984cf78414f54793dea4c)
FetchContent_MakeAvailable(expat)
if(CHALI_NATIVE_STATIC)
    add_library(chali_xml STATIC xml.c)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(chali_xml PRIVATE -ffunction-sections -fdata-sections)
        target_compile_options(expat PRIVATE -ffunction-sections -fdata-sections)
    endif()
else()
    add_library(chali_xml SHARED xml.c)
endif()
target_compile_features(chali_xml PRIVATE c_std_11)
target_link_libraries(chali_xml PRIVATE expat::expat)
set_target_properties(chali_xml PROPERTIES PREFIX "" WINDOWS_EXPORT_ALL_SYMBOLS ON)
install(TARGETS chali_xml RUNTIME DESTINATION . LIBRARY DESTINATION . ARCHIVE DESTINATION .)
install(FILES "${expat_SOURCE_DIR}/COPYING" DESTINATION licenses RENAME expat-COPYING)
