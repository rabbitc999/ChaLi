cmake_minimum_required(VERSION 3.20)
project(chali_desktop_gui LANGUAGES C)
option(CHALI_DESKTOP_AUDIO "Enable the optional audio module" OFF)
option(CHALI_GUI_ICONS "Include raygui built-in icons" OFF)
option(CHALI_GUI_TESTING "Build process-local hidden-window test helpers" OFF)
option(CHALI_NATIVE_STATIC "Build separate native archives for the embedded AOT linker" OFF)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CUSTOMIZE_BUILD ON CACHE BOOL "" FORCE)
set(SUPPORT_MODULE_RAUDIO ${CHALI_DESKTOP_AUDIO} CACHE BOOL "" FORCE)
set(SUPPORT_MODULE_RMODELS OFF CACHE BOOL "" FORCE)
# 窗口与全部控件不需要摄像机、录屏、系统图片剪贴板或三维模型格式。
# 保留 PNG/TTF/FNT 字体路径；测试构建单独启用帧导出与进程内输入回放。
foreach(feature SUPPORT_CAMERA_SYSTEM SUPPORT_CLIPBOARD_IMAGE SUPPORT_GESTURES_SYSTEM
        SUPPORT_MOUSE_GESTURES SUPPORT_GIF_RECORDING SUPPORT_SCREEN_CAPTURE
        SUPPORT_COMPRESSION_API SUPPORT_FILEFORMAT_DDS SUPPORT_FILEFORMAT_GIF
        SUPPORT_FILEFORMAT_QOI SUPPORT_FILEFORMAT_GLTF SUPPORT_FILEFORMAT_IQM
        SUPPORT_FILEFORMAT_M3D SUPPORT_FILEFORMAT_MTL SUPPORT_FILEFORMAT_OBJ
        SUPPORT_FILEFORMAT_VOX SUPPORT_MESH_GENERATION)
    set(${feature} OFF CACHE BOOL "" FORCE)
endforeach()
set(SUPPORT_AUTOMATION_EVENTS ${CHALI_GUI_TESTING} CACHE BOOL "" FORCE)
set(SUPPORT_IMAGE_EXPORT ${CHALI_GUI_TESTING} CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
if(CHALI_GUI_TESTING)
    set(SUPPORT_AUTOMATION_EVENTS ON CACHE BOOL "" FORCE)
    set(SUPPORT_IMAGE_EXPORT ON CACHE BOOL "" FORCE)
endif()
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-ffunction-sections -fdata-sections)
endif()
add_subdirectory("${RAYLIB_SOURCE}" raylib EXCLUDE_FROM_ALL)
if(CHALI_NATIVE_STATIC)
    set(chali_library_kind STATIC)
else()
    set(chali_library_kind SHARED)
endif()
add_library(chali_desktop ${chali_library_kind} desktop.c desktop_window.c desktop_widgets.c desktop_draw.c desktop_texture.c desktop_assets.c)
if(CHALI_GUI_TESTING)
    target_sources(chali_desktop PRIVATE desktop_test.c)
endif()
target_compile_features(chali_desktop PRIVATE c_std_11)
target_compile_definitions(chali_desktop PRIVATE CHALI_GUI_BUILD=1)
target_link_libraries(chali_desktop PRIVATE raylib)
if(NOT WIN32)
    find_package(Threads REQUIRED)
    target_link_libraries(chali_desktop PRIVATE Threads::Threads)
endif()
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    if(APPLE)
        target_link_options(chali_desktop PRIVATE -Wl,-dead_strip)
    else()
        target_link_options(chali_desktop PRIVATE -Wl,--gc-sections)
    endif()
endif()
target_include_directories(chali_desktop PRIVATE "${RAYGUI_SOURCE}/src")
if(NOT CHALI_GUI_ICONS)
    target_compile_definitions(chali_desktop PRIVATE RAYGUI_NO_ICONS)
endif()
if(CHALI_NATIVE_STATIC)
    target_compile_definitions(chali_desktop PRIVATE CHALI_GUI_STATIC=1)
    install(TARGETS raylib ARCHIVE DESTINATION .)
    if(MINGW)
        # 随归档提供只在使用时链接的 Windows 系统导入库。
        foreach(system_library gdi32 winmm)
            execute_process(COMMAND "${CMAKE_C_COMPILER}" "-print-file-name=lib${system_library}.a"
                            OUTPUT_VARIABLE import_library OUTPUT_STRIP_TRAILING_WHITESPACE
                            COMMAND_ERROR_IS_FATAL ANY)
            if(NOT IS_ABSOLUTE "${import_library}" OR NOT EXISTS "${import_library}")
                message(FATAL_ERROR "Cannot locate lib${system_library}.a")
            endif()
            install(FILES "${import_library}" DESTINATION lib)
        endforeach()
    endif()
    if(CHALI_DESKTOP_AUDIO)
        add_library(chali_audio STATIC audio.c)
        target_compile_features(chali_audio PRIVATE c_std_11)
        target_compile_definitions(chali_audio PRIVATE CHALI_GUI_STATIC=1)
        target_link_libraries(chali_audio PRIVATE raylib)
        install(TARGETS chali_audio ARCHIVE DESTINATION .)
    endif()
elseif(CHALI_DESKTOP_AUDIO)
    target_sources(chali_desktop PRIVATE audio.c)
endif()
set_target_properties(chali_desktop PROPERTIES PREFIX "" WINDOWS_EXPORT_ALL_SYMBOLS OFF)
install(TARGETS chali_desktop RUNTIME DESTINATION . LIBRARY DESTINATION . ARCHIVE DESTINATION .)
install(FILES desktop.h desktop_draw.h controls.def DESTINATION include)
install(FILES "${RAYLIB_SOURCE}/LICENSE" DESTINATION licenses RENAME raylib-LICENSE)
install(FILES "${RAYGUI_SOURCE}/LICENSE" DESTINATION licenses RENAME raygui-LICENSE)
