cmake_minimum_required(VERSION 3.15)
project(ChaLiAst LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(parseengine_ast STATIC ast.cpp ast_type.cpp ast_node.cpp ast_data.cpp aggregate_optimize.cpp)
target_include_directories(parseengine_ast PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(MSVC)
  target_compile_options(parseengine_ast PRIVATE /W4 /utf-8)
else()
  target_compile_options(parseengine_ast PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(ast-cli main.cpp)
target_link_libraries(ast-cli PRIVATE parseengine_ast)
enable_testing()
add_executable(ast-tests test_ast.cpp)
target_link_libraries(ast-tests PRIVATE parseengine_ast)
add_test(NAME ast-tests COMMAND ast-tests)

target_compile_definitions(parseengine_ast PUBLIC CHALI_CUSTOM_ALLOCATOR=1)
