cmake_minimum_required(VERSION 3.15)
project(ChaLiNativeRuntime LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
    add_compile_options(/utf-8)
endif()

option(RUNTIME_WITH_REFLECTION "启用注解与反射运行时支持" ON)
option(RUNTIME_WITH_PANIC_TRACE "Retain panic call stacks" ON)
option(RUNTIME_WITH_THREADS "编入宿主线程和共享值" ON)
option(RUNTIME_WITH_DEFAULT_HEAP "编入默认动态字节堆" ON)
option(RUNTIME_WITH_HOST_EXCEPTION_GUARD "保护宿主回调异常" ON)
option(RUNTIME_WITH_HOST_IO "编入按路径加载与保存" ON)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/aot_sources.txt")
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/aot_sources.txt" RUNTIME_CORE_SOURCES)
add_library(chali_stackvm STATIC ${RUNTIME_CORE_SOURCES})
target_include_directories(chali_stackvm PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/..)
foreach(feature THREADS DEFAULT_HEAP HOST_EXCEPTION_GUARD HOST_IO)
    target_compile_definitions(chali_stackvm PUBLIC RUNTIME_WITH_${feature}=$<BOOL:${RUNTIME_WITH_${feature}}>)
endforeach()
target_compile_definitions(chali_stackvm PUBLIC RUNTIME_WITH_REFLECTION=$<BOOL:${RUNTIME_WITH_REFLECTION}>)
target_compile_definitions(chali_stackvm PUBLIC RUNTIME_WITH_PANIC_TRACE=$<BOOL:${RUNTIME_WITH_PANIC_TRACE}>)
add_subdirectory(../../source/abi ${CMAKE_CURRENT_BINARY_DIR}/standard-records)

# 只构建原生 BUILD 共享运行时服务。
target_compile_definitions(chali_stackvm PUBLIC CHALI_CUSTOM_ALLOCATOR=1 RUNTIME_BUILD_ONLY=1)
add_library(chali_native_core ALIAS chali_stackvm)
